country_event = {
	id = btk_squad.50
	title = btk_squad.50.t
	desc = btk_squad.50.d
	picture = GFX_report_event_BTK_89squad_decay
	
	is_triggered_only = no
	
	trigger = {
		original_tag = BTK
		has_country_flag = BTK_narezki_factions_unlocked
		# At least one faction is not resolved
		OR = {
			AND = {
				NOT = { has_country_flag = BTK_rolfs_integrated_flag }
				NOT = { has_country_flag = BTK_rolfs_destroyed_flag }
			}
			AND = {
				NOT = { has_country_flag = BTK_mbrff_integrated_flag }
				NOT = { has_country_flag = BTK_mbrff_destroyed_flag }
			}
			AND = {
				NOT = { has_country_flag = BTK_zar_integrated_flag }
				NOT = { has_country_flag = BTK_zar_destroyed_flag }
			}
		}
		NOT = { has_country_flag = BTK_narezki_event_cooldown }
	}

	mean_time_to_happen = {
		days = 80
		modifier = {
			factor = 0.8
			has_stability < 0.3
		}
		modifier = {
			factor = 2
			has_stability > 0.6
		}
	}
	
	option = {
		name = btk_squad.50.a
		ai_chance = {
			factor = 100
		}
		
		add_stability = -0.03
		
		hidden_effect = {
			set_country_flag = { flag = BTK_narezki_event_cooldown days = 50 value = 1 }
		}
	}
}
