# Event for arresting the coordinator
country_event = {
	id = chb_sever.2
	title = chb_sever.2.t
	desc = chb_sever.2.d
	picture = GFX_report_event_generic_factory
	hidden = yes

	is_triggered_only = yes

	immediate = {
		# Cycle the coordinator frame variable
		if = {
			limit = { check_variable = { SeverCoordinator = 1 } }
			set_variable = { SeverCoordinator = 2 }
		}
		else_if = {
			limit = { check_variable = { SeverCoordinator = 2 } }
			set_variable = { SeverCoordinator = 3 }
		}
		else = {
			set_variable = { SeverCoordinator = 1 }
		}
		
		# Give a temporary buff to resistance target
		add_ideas = chb_coordinator_arrested_buff
	}
}

#SEVER AUTOSTART
country_event = {
	id = chb_sever.7
	title = chb_sever.7.t
	desc = chb_sever.7.d
	picture = GFX_report_event_generic_siloviki

	is_triggered_only = yes

	option = {
		name = chb_sever.7.a #yes
		ai_chance = {
			factor = 30
		}
		if = {
			limit = {
				NOT = {
					has_completed_focus = chb_northern_problems
				}
			}
			complete_focus = chb_northern_problems
		}
		
		every_owned_state = {
			limit = { is_core_of = NCB }
			add_dynamic_modifier = { modifier = chb_severe_separatism_modifier }
		}
	}
}
