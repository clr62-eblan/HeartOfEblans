# Initialize all BTK narezki factions variables
BTK_initialize_narezki_factions = {
	# Rolfs - higher starting loyalty
	set_variable = { BTK_rolfs_loyalty = 35 }
	set_variable = { BTK_rolfs_popularity = 25 }
	
	# MBRFF - low loyalty
	set_variable = { BTK_mbrff_loyalty = 15 }
	set_variable = { BTK_mbrff_popularity = 30 }
	
	# Zar - low loyalty
	set_variable = { BTK_zar_loyalty = 10 }

	# Unlock factions system
	set_country_flag = BTK_narezki_factions_unlocked
}
