CHECK_IS_SHPANA_FULLY_CONNECTED = {
    if = {
        limit = {
            check_variable = { SHPANA_CONNECT_STATUS = 4 }
            NOT = { has_country_flag = SHPANA_REUNIFIED_FULLY }
        }
        set_cosmetic_tag = KUS_SHPANA
        set_country_flag = SHPANA_REUNIFIED_FULLY
    }
}
